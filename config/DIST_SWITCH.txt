hostname DIST_SWITCH
no ip domain-lookup
ip routing

! --- Cấu hình VLAN ---
vlan 10
 name CLIENT_VLAN
vlan 20
 name SERVER_VLAN
vlan 99
 name MGMT_NOC
exit

! --- STP Tuning: Làm Root Bridge cho các VLAN ---
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,99 root primary

! --- Cấu hình Cổng đấu nối lên CORE_ROUTER (Routed Port) ---
interface Ethernet0/0
 description UPLINK-TO-CORE_ROUTER
 no switchport
 ip address 10.0.0.2 255.255.255.252
 no shutdown
exit

! --- Cấu hình SVIs (Gateway cho các VLAN) ---
interface Vlan10
 description GW-CLIENTS-ATTACKER-VICTIM
 ip address 192.168.10.1 255.255.255.0
 no shutdown
exit

interface Vlan20
 description GW-SERVERS
 ip address 192.168.20.1 255.255.255.0
 no shutdown
exit

interface Vlan99
 description GW-MANAGEMENT
 ip address 10.10.99.2 255.255.255.0
 no shutdown
exit

! --- Cấu hình Trunk Link xuống ACCESS_SWITCH ---
interface Ethernet0/1
 description TRUNK-DOWN-TO-ACCESS_SWITCH
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 no shutdown
exit

! --- Định tuyến OSPF kết nối CORE_ROUTER ---
router ospf 1
 router-id 2.2.2.2
 network 10.0.0.0 0.0.0.3 area 0
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 passive-interface Vlan10
 passive-interface Vlan20
exit



! Gửi log về NOC server
logging host 10.10.99.10
logging trap informational
logging source-interface Vlan99
service timestamps log datetime msec localtime show-timezone
